/** Strict sanitizer: only p/div/br/b/i/ul/ol/li and <img data-i="ID"> survive; everything else is dropped, text is escaped. */
function vrSanitize_(html) {
  const ok = { p: 1, div: 1, br: 1, b: 1, i: 1, ul: 1, ol: 1, li: 1 };
  const esc = function (t) { return t.replace(/&(?!(amp|lt|gt|quot|#39|nbsp);)/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); };
  html = String(html == null ? '' : html); let out = '', last = 0, m; const re = /<(\/?)([a-zA-Z][a-zA-Z0-9]*)([^<>]*)>/g;
  while ((m = re.exec(html))) {
    out += esc(html.slice(last, m.index)); last = re.lastIndex;
    const tag = m[2].toLowerCase();
    if (tag === 'img') { const id = /data-i="([A-Za-z0-9_-]{10,80})"/.exec(m[3]); if (id && !m[1]) out += '<img data-i="' + id[1] + '">'; }
    else if (ok[tag]) out += m[1] ? (tag === 'br' ? '' : '</' + tag + '>') : (tag === 'br' ? '<br>' : '<' + tag + '>');
  }
  return out + esc(html.slice(last));
}
/** One-off helper: turn a Google Doc (text + inline images) into ONE submitted comment, keeping the original date. Images are copied into the shots folder; the Doc is left untouched. */
function vrImportDocAsComment_(docId, key, v, by, at) {
  const body = DocumentApp.openById(docId).getBody(), ET = DocumentApp.ElementType, parts = [];
  const esc = function (t) { return String(t).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); };
  function walk(el) {
    for (let j = 0; j < el.getNumChildren(); j++) {
      const c = el.getChild(j), t = c.getType();
      if (t === ET.INLINE_IMAGE) { const f = vrShots_().createFile(c.asInlineImage().getBlob().setName(String(key).replace(/[|\s]/g, '_') + '_v' + v + '_import_' + Utilities.getUuid().slice(0, 6))); parts.push('<img data-i="' + f.getId() + '">'); }
      else if (t === ET.TEXT) { const tx = c.asText().getText(); if (tx.trim()) parts.push('<p>' + esc(tx) + '</p>'); }
      else if (c.getNumChildren) walk(c);
    }
  }
  walk(body);
  const html = vrSanitize_(parts.join('')); if (!html) return '';
  const id = 'c' + Utilities.getUuid().replace(/-/g, '').slice(0, 8);
  vrTab_('VideoComments', true).appendRow([id, key, v, by, html, at || vrNow_()]); vrLog_('import_doc_comment', key, docId, by + ' -> ' + id);
  return id;
}
function vrJson_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function doPost(e) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) return vrJson_({ ok: false, code: 'busy', message: 'Try again in a moment.' });
  try { return vrPost_(JSON.parse((e && e.postData && e.postData.contents) || '{}')); }
  catch (err) { return vrJson_({ ok: false, code: 'write_failed', message: String(err) }); }
  finally { lock.releaseLock(); }
}

/** create=false (GET path) never creates a sheet, so reads cannot race. All columns are plain text (no =formula / number mangling). */
function vrTab_(name, create) {
  const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(name);
  if (!sh && create) {
    sh = ss.insertSheet(name); sh.appendRow(VR_HEAD[name]); sh.setFrozenRows(1);
    sh.getRange(1, 1, sh.getMaxRows(), VR_HEAD[name].length).setNumberFormat('@');
  }
  return sh;
}
function vrRows_(name) {
  const sh = vrTab_(name, false); if (!sh) return [];
  const v = sh.getDataRange().getValues(); const h = v.shift();
  return v.map(function (r, i) { const o = { _row: i + 2 }; h.forEach(function (k, j) { o[k] = r[j]; }); return o; });
}
function vrNow_(d) { return Utilities.formatDate(d || new Date(), 'Etc/UTC', 'yyyy-MM-dd HH:mm') + ' UTC'; }
function vrBy_(s) { return String(s || '').replace(/[\r\n\t]+/g, ' ').trim().slice(0, 60); }
function vrLog_(action, key, file, detail) { vrTab_('VideoLog', true).appendRow([vrNow_(), action, key || '', file || '', detail || '']); }
function vrMigrateComments_() {                              // older sheet: [key, v, by, html, updatedAt] -> one row per comment with an id
  const sh = vrTab_('VideoComments', false); if (!sh || String(sh.getRange(1, 1).getValue()) !== 'key') return;

  sh.getRange(1, 1, sh.getMaxRows(), VR_HEAD.VideoComments.length).setNumberFormat('@');
  old.forEach(function (r) { sh.appendRow(['c' + Utilities.getUuid().replace(/-/g, '').slice(0, 8), r.key, r.v, r.by, r.html, r.updatedAt || vrNow_()]); });
}
function vrInit_() {

}

function vrSub_(parent, name, create) { const it = parent.getFoldersByName(name); if (it.hasNext()) return it.next(); return create ? parent.createFolder(name) : null; }
function vrInbox_() {
  const id = PropertiesService.getScriptProperties().getProperty('VR_INBOX');
  if (id) return DriveApp.getFolderById(id);
  let f = DriveApp.getRootFolder();
  for (let i = 0; i < VR_INBOX_PATH.length && f; i++) f = vrSub_(f, VR_INBOX_PATH[i], false);
  return f;
}
/** free file name inside a folder: never overwrite -> add _2, _3 ... */
function vrFreeName_(folder, name) {
  const m = /^(.*?)(\.[^.]*)?$/.exec(name); let n = name, i = 2;
  while (folder.getFilesByName(n).hasNext()) { n = m[1] + '_' + i + (m[2] || ''); i++; }
  return n;
}
/** move FIRST, then rename (a failed move leaves the file untouched; a failed rename leaves it safely in the target folder). Name is checked free in the target before the move. */
function vrMoveInto_(file, folder, newName) {
  const free = vrFreeName_(folder, newName); file.moveTo(folder);
  try { file.setName(free); } catch (err) { vrLog_('rename_failed', '', file.getId(), String(err)); }
}
function vrCodes_(name) {                                   // unique video codes found in a file name, e.g. "MS01 (1).mp4" -> ['MS01']
  const base = String(name).replace(/\.[^.]*$/, '').toUpperCase(), re = /(^|[^A-Z0-9])([A-Z]{1,3}\d{2})(?![A-Z0-9])/g, out = []; let m;
  while ((m = re.exec(base))) if (out.indexOf(m[2]) < 0 && !/^V\d\d$/.test(m[2])) out.push(m[2]);   // _v12 is a version tag, not a code
  return out;
}
function vrVersions_() {                                    // key -> [{v,fileId,origName,size,date}] sorted
  const out = {};
  vrRows_('VideoVersions').forEach(function (r) { (out[r.key] = out[r.key] || []).push({ v: Number(r.v), id: r.fileId, name: r.origName, size: Number(r.size), date: r.date, doc: r.docId || '' }); });
  Object.keys(out).forEach(function (k) { out[k].sort(function (a, b) { return a.v - b.v; }); });
  return out;
}
function vrAddVersion_(key, file, origName) {
  const vs = vrVersions_()[key] || [], v = vs.length ? vs[vs.length - 1].v + 1 : 1;
  const prev = vs.length ? vs[vs.length - 1].doc : '';
  let doc = ''; try { doc = vrMakeDoc_(key, v, prev); } catch (err) { vrLog_('doc_failed', key, '', String(err)); }
  vrTab_('VideoVersions', true).appendRow([key, v, file.getId(), origName, file.getSize(), vrNow_(file.getLastUpdated()), doc]);
  return v;
}

/** Doc "<PILLAR> <LANG> <CODE> v<n> - changes": 3-column table Timecode | Screenshot | Change, anyone with the link can EDIT. */
function vrMakeDoc_(key, v, prevDocId, label) {
  const a = key.split('|'), doc = DocumentApp.create(a[0] + ' ' + a[1] + ' ' + a[2] + ' v' + v + ' - ' + (label === 'ALL' ? 'all changes' : label ? 'comments by ' + label : 'changes'));   // blank doc: reviewers add their own screenshots/timecodes/notes
  doc.saveAndClose();
  const f = DriveApp.getFileById(doc.getId()); f.moveTo(vrDocsFolder_());
  f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.EDIT);

/** make sure every version has a doc (run by hand once for existing files; also runs inside the trigger) */
function vrBackfillDocs() {
  vrInit_(); const sh = vrTab_('VideoVersions', true);
  vrRows_('VideoVersions').forEach(function (r) {
    if (r.docId) return;
    const vs = vrVersions_()[r.key] || [], prev = vs.filter(function (x) { return x.v === Number(r.v) - 1; })[0];
    try { sh.getRange(r._row, 7).setValue(vrMakeDoc_(r.key, Number(r.v), prev ? prev.doc : '')); } catch (err) { vrLog_('doc_failed', r.key, '', String(err)); }
  });
}

/** The editor sees ONE doc: "<...> v<n> - all changes" = every finished reviewer doc in order, each under a heading with the name + date.
 *  Rebuilt from scratch (copy of the reviewers' elements) whenever someone presses Changes done, and by the page's Refresh link. Reviewer docs are never modified. */
function vrCombinedId_(key, v, create) {
  const rows = vrRows_('VideoDocs').filter(function (d) { return d.key === key && Number(d.v) === v && d.by === 'ALL'; });
  if (rows.length) return rows[0].docId;
  if (!create) return '';
  const id = vrMakeDoc_(key, v, '', 'ALL'); vrTab_('VideoDocs', true).appendRow([key, v, id, 'ALL', vrNow_()]); return id;
}
function vrRebuildCombined_(key, v) {
  const vs = vrVersions_()[key] || [], rec = vs.filter(function (x) { return x.v === v; })[0]; if (!rec) return '';
  const done = vrRows_('VideoEvents').filter(function (e) { return e.key === key && Number(e.v) === v && e.type === 'changes' && e.docId; });
  if (!done.length) return '';
  const id = vrCombinedId_(key, v, true), doc = DocumentApp.openById(id), body = doc.getBody(), ET = DocumentApp.ElementType;
  body.clear();
  const first = body.getNumChildren() ? body.getChild(0) : null;
  done.forEach(function (e, n) {                                   // done rows are in sheet order = order the reviewers finished
    body.appendParagraph('Comments by ' + e.by + ' - ' + e.at).setHeading(DocumentApp.ParagraphHeading.HEADING2);
    let src; try { src = DocumentApp.openById(e.docId).getBody(); } catch (err) { body.appendParagraph('(could not open ' + e.by + "'s document: " + err + ')'); return; }
    for (let i = 0; i < src.getNumChildren(); i++) {
      const el = src.getChild(i), t = el.getType();
      try {
        if (t === ET.PARAGRAPH) body.appendParagraph(el.copy());
        else if (t === ET.LIST_ITEM) body.appendListItem(el.copy());
        else if (t === ET.TABLE) body.appendTable(el.copy());
        else if (t === ET.HORIZONTAL_RULE) body.appendHorizontalRule();
        else if (t !== ET.PAGE_BREAK) body.appendParagraph('(an item of type ' + t + ' was not copied - see ' + e.by + "'s own document)");
      } catch (err) { body.appendParagraph('(an item could not be copied: ' + err + ')'); }
    }
  });
  doc.saveAndClose(); vrLog_('combined_rebuilt', key, id, 'v' + v + ', ' + done.length + ' reviewer doc(s)');
  return id;
}

// ---------- the 15-minute trigger ----------
/** Run vrInstallTrigger() ONCE by hand after deploying. */
function vrInstallTrigger() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'vrIngestAll') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('vrIngestAll').timeBased().everyMinutes(15).create();
}
function vrIngestAll() {
  const lock = LockService.getScriptLock(); if (!lock.tryLock(25000)) return;
  try {
    vrInit_();
    const inbox = vrInbox_(), unclear = []; if (!inbox) throw new Error('For Revision Renders folder not found');
    VR_PILLARS.forEach(function (p) {
      const pf = vrSub_(inbox, p, false); if (!pf) return;
      const lfs = pf.getFolders();
      while (lfs.hasNext()) {
        const L = lfs.next(), lang = L.getName(); if (lang.charAt(0) === '_') continue;
        try { vrIngestFolder_(L, p, lang, unclear); } catch (err) { vrLog_('error', p + '|' + lang, '', String(err)); }
      }
    });
    vrBackfillDocs();
    PropertiesService.getScriptProperties().setProperty('VR_UNCLEAR', JSON.stringify(unclear.slice(0, 40)));
    CacheService.getScriptCache().remove('vr_finals');
  } catch (err) { try { vrLog_('error', '', '', String(err)); } catch (_) {} }
  finally { lock.releaseLock(); }
}
function vrIngestFolder_(L, p, lang, unclear, noSettle) {
  const todo = [], now = Date.now(), fs = L.getFiles(), exact = {}, vers = vrVersions_();
  const settled = function (f) { return noSettle || now - f.getLastUpdated().getTime() >= VR_SETTLE_MS; };   // noSettle: the editor pressed Uploaded, so the file is complete
  while (fs.hasNext()) {
    const f = fs.next(), name = f.getName(), ex = /^([A-Z]{1,3}\d{2})\.mp4$/.exec(name);
    if (ex) { (exact[ex[1]] = exact[ex[1]] || []).push(f); continue; }
    if (!settled(f)) continue;                                                     // still uploading: next run
    const codes = vrCodes_(name), isMp4 = /\.mp4$/i.test(name);
    if (!isMp4 || codes.length !== 1) { unclear.push({ pillar: p, lang: lang, name: name, why: !isMp4 ? 'not an .mp4' : codes.length ? 'several codes in the name' : 'no code in the name' }); continue; }
    todo.push({ f: f, code: codes[0], name: name, t: f.getLastUpdated().getTime() });
  }
  const cur = {};
  Object.keys(exact).forEach(function (code) {
    const key = p + '|' + lang + '|' + code, vs = vers[key] || [], lastId = vs.length ? vs[vs.length - 1].id : '';
    let list = exact[code].slice().sort(function (a, b) { return b.getLastUpdated().getTime() - a.getLastUpdated().getTime(); });   // newest first
    const reg = list.filter(function (f) { return f.getId() === lastId; })[0] || list[0];
    list.forEach(function (f) { if (f !== reg) unclear.push({ pillar: p, lang: lang, name: f.getName(), why: 'second file called ' + code + '.mp4 (left untouched)' }); });
    if (!vs.some(function (x) { return x.id === reg.getId(); }) && !settled(reg)) return;   // a fresh hand-placed <CODE>.mp4 waits too
    cur[code] = reg;
    if (!vs.some(function (x) { return x.id === reg.getId(); })) vrAddVersion_(key, reg, reg.getName());   // hand-placed file becomes a version
  });
  todo.sort(function (a, b) { return a.t - b.t; });                                  // oldest first: later upload = newest version
  todo.forEach(function (t) {
    const key = p + '|' + lang + '|' + t.code, old = cur[t.code], pre = VR_PREFIX[p];
    if (pre && t.code.replace(/\d+$/, '') !== pre) vrLog_('warn_code_pillar', key, t.f.getId(), 'code ' + t.code + ' in a ' + p + ' folder (expected ' + pre + '##) - ingested anyway');
    if (old) {                                                                       // previous version -> _old with its number + date (move first)
      const vs = vrVersions_()[key] || [], n = vs.length ? vs[vs.length - 1].v : 1;
      const dst = vrSub_(L, '_old', true), to = t.code + '_v' + n + '_' + Utilities.formatDate(old.getLastUpdated(), 'Etc/UTC', 'yyyy-MM-dd_HHmm') + '.mp4';
      vrMoveInto_(old, dst, to); vrLog_('to_old', key, old.getId(), to); delete cur[t.code];
    }
    t.f.setName(t.code + '.mp4'); cur[t.code] = t.f;
    const v = vrAddVersion_(key, t.f, t.name);
    vrLog_('new_version', key, t.f.getId(), 'v' + v + ' from "' + t.name + '"');
  });
}

// ---------- reads ----------
/** finals = <CODE>.mp4 files in NEW NOW MATERIALS/<PILLAR>/VIDEO/<LANG>/ (read-only scan, cached 5 min) */
function vrFinals_() {
  const cache = CacheService.getScriptCache(), hit = cache.get('vr_finals'); if (hit) return JSON.parse(hit);
  const out = [], root = DriveApp.getFolderById(VR_ROOT);
  VR_PILLARS.forEach(function (p) {
    const pf = vrSub_(root, p, false), vf = pf && vrSub_(pf, 'VIDEO', false); if (!vf) return;
    const lf = vf.getFolders();
    while (lf.hasNext()) {
      const L = lf.next(), fs = L.getFiles();
      while (fs.hasNext()) { const f = fs.next(), m = /^([A-Z]{1,3}\d{2})\.mp4$/.exec(f.getName()); if (m) out.push({ pillar: p, lang: L.getName(), code: m[1], id: f.getId() }); }
    }
  });
  const txt = JSON.stringify(out); if (txt.length < 95000) cache.put('vr_finals', txt, 300);
  return out;
}
function vrTree_() {
  const vers = vrVersions_(), items = {};
  Object.keys(vers).forEach(function (k) { const a = k.split('|'); items[k] = { pillar: a[0], lang: a[1], code: a[2], final: null, versions: vers[k] }; });
  vrFinals_().forEach(function (f) {
    const k = f.pillar + '|' + f.lang + '|' + f.code;
    items[k] = items[k] || { pillar: f.pillar, lang: f.lang, code: f.code, final: null, versions: [] }; items[k].final = { id: f.id };
  });
  return Object.keys(items).map(function (k) { return items[k]; });
}
function vrGet_(e) {
  try {
    const a = e.parameter.action;
    if (a === 'v_all') {
      const notes = vrRows_('VideoNotes').map(function (n) {
        return { id: n.id, key: n.key, v: Number(n.v), tc: Number(n.tc), tcText: n.tcText, text: n.text, by: n.by, at: n.at, shotId: n.shotId, fixed: n.fixed === 'yes', fixedBy: n.fixedBy };
      });
      let unclear = []; try { unclear = JSON.parse(PropertiesService.getScriptProperties().getProperty('VR_UNCLEAR') || '[]'); } catch (_) {}
      return vrJson_({ ok: true, apiKey: PropertiesService.getScriptProperties().getProperty('VR_API_KEY') || '', expected: VR_EXPECTED, langs: VR_LANGS, tree: vrTree_(), notes: notes, events: vrRows_('VideoEvents').map(function (e) { return { key: e.key, v: Number(e.v), type: e.type, by: e.by, at: e.at, docId: e.docId || '' }; }),
        comments: vrRows_('VideoComments').map(function (c) { return { id: c.id, key: c.key, v: Number(c.v), by: c.by, html: c.html, at: c.at }; }),
        docs: vrRows_('VideoDocs').map(function (d) { return { key: d.key, v: Number(d.v), docId: d.docId, by: d.by, at: d.at }; }), unclear: unclear });
    }
    if (a === 'v_shot') {                                  // only files inside our shots folder can be fetched
      const f = DriveApp.getFileById(String(e.parameter.id || '')), par = f.getParents();
      if (!par.hasNext() || par.next().getId() !== vrShots_().getId()) return vrJson_({ ok: false, code: 'bad_shot' });
      return vrJson_({ ok: true, data: 'data:image/jpeg;base64,' + Utilities.base64Encode(f.getBlob().getBytes()) });
    }
    return vrJson_({ ok: false, code: 'bad_action' });
  } catch (err) { return vrJson_({ ok: false, code: 'read_failed', message: String(err) }); }
}
function vrShots_() {
  const props = PropertiesService.getScriptProperties(), id = props.getProperty('VR_SHOTS');
  if (id) { try { return DriveApp.getFolderById(id); } catch (_) {} }
  const par = DriveApp.getFileById(SpreadsheetApp.getActiveSpreadsheet().getId()).getParents();
  const f = (par.hasNext() ? par.next() : DriveApp.getRootFolder()).createFolder('NOW VIDEO REVIEW SHOTS');
  props.setProperty('VR_SHOTS', f.getId()); return f;
}

// ---------- writes (called from doPost, which already holds the script lock) ----------
function vrPinOk_(pin) {
  const cache = CacheService.getScriptCache(), n = Number(cache.get('vr_pinfail') || 0);
  if (n >= 5) return 'locked';
  const want = PropertiesService.getScriptProperties().getProperty('VR_PIN');
  if (want && String(pin) === want) return 'ok';
  cache.put('vr_pinfail', String(n + 1), 600); return 'bad';
}
function vrPost_(p) {
  try {
    vrInit_();
    const by = vrBy_(p.by);
    if (p.action === 'v_approve') {                          // only Sarosh's PIN action; copies the file, never moves the review copy
      const chk = vrPinOk_(p.pin);
      if (chk === 'locked') return vrJson_({ ok: false, code: 'locked', message: 'Too many wrong PINs. Try again in 10 minutes.' });
      if (chk !== 'ok') return vrJson_({ ok: false, code: 'bad_pin', message: 'Wrong PIN.' });
      return vrApprove_(String(p.key), Number(p.v));
    }
    if (p.action === 'v_farah_ok') {                         // English only, no PIN/name: Farah approves a version the reviewers approved -> goes to Sarosh
      const key = String(p.key), v = Number(p.v), vs = vrVersions_()[key] || [], ev = vrRows_('VideoEvents').filter(function (e) { return e.key === key; }).pop();
      if (key.split('|')[1] !== 'ENGLISH') return vrJson_({ ok: false, code: 'not_english', message: 'Farah checks English videos only.' });
      if (!vs.length || v !== vs[vs.length - 1].v) return vrJson_({ ok: false, code: 'stale', message: 'A newer version arrived - reload the page.' });
      const lastCh = vrRows_('VideoEvents').filter(function (e) { return e.key === key && (e.type === 'changes' || e.type === 'farah_changes'); }).pop();
      const backToFarah = ev && ev.type === 'resubmitted' && Number(ev.v) === v && lastCh && lastCh.type === 'farah_changes';   // Farah asked for changes -> the fixed version skips the reviewers
      if (!ev || Number(ev.v) !== v || (ev.type !== 'ok' && !backToFarah)) return vrJson_({ ok: false, code: 'not_ok', message: 'This version is not waiting for Farah.' });
      vrTab_('VideoEvents', true).appendRow([key, v, 'farah_ok', 'Farah', vrNow_(), '']); return vrJson_({ ok: true });
    }
    if (p.action === 'v_reopen') {                           // PIN: send an approved video back to review (event only; final copy, history and comments stay)
      const chk = vrPinOk_(p.pin);
      if (chk === 'locked') return vrJson_({ ok: false, code: 'locked', message: 'Too many wrong PINs. Try again in 10 minutes.' });
      if (chk !== 'ok') return vrJson_({ ok: false, code: 'bad_pin', message: 'Wrong PIN.' });
      const key = String(p.key), vs = vrVersions_()[key] || [], last = vs[vs.length - 1];
      const ev = vrRows_('VideoEvents').filter(function (e) { return e.key === key && (e.type === 'approved' || e.type === 'reopened'); }).pop();
      if (!last || !ev || ev.type !== 'approved') return vrJson_({ ok: false, code: 'not_approved', message: 'This video is not in the approved state.' });
      vrTab_('VideoEvents', true).appendRow([key, last.v, 'reopened', 'Sarosh', vrNow_(), '']); vrLog_('reopened', key, last.id, 'sent back to review');
      return vrJson_({ ok: true });
    }
    if (p.action === 'v_refresh') {                          // rebuild the editor's combined doc
      const v = Number(p.v); try { vrRebuildCombined_(String(p.key), v); } catch (err) { return vrJson_({ ok: false, code: 'rebuild_failed', message: String(err) }); }
      return vrJson_({ ok: true });
    }
    if (p.action === 'v_uploaded') {                         // editor presses Uploaded: ingest THIS folder now, then tag the upload
      const key = String(p.key), a = key.split('|');
      if (!String(p.by || '').trim()) return vrJson_({ ok: false, code: 'no_name', message: 'Please type your name first.' });
      if (a.length !== 3 || !VR_EXPECTED.hasOwnProperty(a[0]) || VR_LANGS.indexOf(a[1]) < 0) return vrJson_({ ok: false, code: 'bad_key' });
      const inbox = vrInbox_(), pf = inbox && vrSub_(inbox, a[0], false), L = pf && vrSub_(pf, a[1], false), unclear = [];
      if (L) vrIngestFolder_(L, a[0], a[1], unclear, true);
      const vs = vrVersions_()[key] || [];
      if (!vs.length) {
        const hint = unclear.length ? ' Files without a usable code: ' + unclear.map(function (u) { return u.name + ' (' + u.why + ')'; }).join(', ') + '.' : '';
        return vrJson_({ ok: false, code: 'no_file', message: 'File not found yet in For Revision Renders/' + a[0] + '/' + a[1] + '/ - upload it first (name must contain ' + a[2] + ').' + hint });
      }
      const v = vs[vs.length - 1].v;
      const tagged = vrRows_('VideoEvents').some(function (e) { return e.key === key && Number(e.v) === v && (e.type === 'uploaded' || e.type === 'resubmitted'); });
      if (!tagged) vrTab_('VideoEvents', true).appendRow([key, v, 'uploaded', vrBy_(p.by), vrNow_(), '']);
      return vrJson_({ ok: true, v: v });
    }
    if (p.action === 'v_comment_add') {                      // "Submit comment": one new item
      const key = String(p.key), v = Number(p.v), vs = vrVersions_()[key] || [];
      if (!by) return vrJson_({ ok: false, code: 'no_name', message: 'Please type your name first.' });
      if (!vs.length || v !== vs[vs.length - 1].v) return vrJson_({ ok: false, code: 'stale', message: 'A newer version arrived - reload the page (your text is still in the box).' });
      const html = vrSanitize_(p.html); if (!/\S/.test(html.replace(/<[^>]*>|&nbsp;/g, '').replace(/\s/g, '')) && !/<img/.test(html)) return vrJson_({ ok: false, code: 'empty', message: 'Write something or paste a screenshot first.' });
      if (html.length > 150000) return vrJson_({ ok: false, code: 'too_big', message: 'Comment too long.' });
      const id = 'c' + Utilities.getUuid().replace(/-/g, '').slice(0, 8), at = vrNow_();
      vrTab_('VideoComments', true).appendRow([id, key, v, by, html, at]); return vrJson_({ ok: true, id: id, at: at });
    }
    if (p.action === 'v_comment_edit' || p.action === 'v_comment_delete') {   // only by re-typing the same name
      const sh = vrTab_('VideoComments', true), row = vrRows_('VideoComments').filter(function (c) { return c.id === String(p.id); })[0];
      if (!by) return vrJson_({ ok: false, code: 'no_name', message: 'Please type your name first.' });
      if (!row) return vrJson_({ ok: false, code: 'gone', message: 'That comment no longer exists.' });
      if (String(row.by).toLowerCase() !== by.toLowerCase()) return vrJson_({ ok: false, code: 'not_yours', message: 'Only ' + row.by + ' can change this comment (type the same name).' });
      if (p.action === 'v_comment_delete') { sh.deleteRow(row._row); return vrJson_({ ok: true }); }
      const html = vrSanitize_(p.html); if (!/<img/.test(html) && !html.replace(/<[^>]*>|&nbsp;/g, '').replace(/\s/g, '')) return vrJson_({ ok: false, code: 'empty', message: 'A comment cannot be empty - delete it instead.' });
      sh.getRange(row._row, 5).setValue(html); return vrJson_({ ok: true });
    }
    if (p.action === 'v_image') {                            // a pasted screenshot -> shots folder, returns the id
      if (!by) return vrJson_({ ok: false, code: 'no_name', message: 'Please type your name first.' });
      const m = /^data:image\/jpeg;base64,([A-Za-z0-9+\/=]+)$/.exec(String(p.image || ''));
      if (!m || m[1].length > 3000000) return vrJson_({ ok: false, code: 'bad_image', message: 'Image rejected (JPEG up to about 2 MB).' });
      const f = vrShots_().createFile(Utilities.newBlob(Utilities.base64Decode(m[1]), 'image/jpeg', String(p.key).replace(/[|\s]/g, '_') + '_v' + Number(p.v) + '_' + Utilities.getUuid().slice(0, 6) + '.jpg'));
      return vrJson_({ ok: true, id: f.getId() });
    }
    if (p.action === 'v_doc') {                              // create the version's doc on first open if the trigger has not yet
      const vs = vrVersions_()[String(p.key)] || [], rec = vs.filter(function (x) { return x.v === Number(p.v); })[0];
      if (!rec) return vrJson_({ ok: false, code: 'bad_version' });
      if (rec.doc) return vrJson_({ ok: true, docId: rec.doc });
      vrBackfillDocs(); return vrJson_({ ok: true });
    }
    if (!by) return vrJson_({ ok: false, code: 'no_name', message: 'Please type your name first.' });
    if (p.action === 'v_event') {                            // name tag + date for every workflow step
      const t = String(p.type), v = Number(p.v), key = String(p.key), vs = vrVersions_()[key] || [];
      if (['uploaded', 'resubmitted', 'changes', 'ok', 'farah_changes'].indexOf(t) < 0 || (t === 'farah_changes' && key.split('|')[1] !== 'ENGLISH') || !vs.length || v !== vs[vs.length - 1].v) return vrJson_({ ok: false, code: 'stale', message: 'A newer version arrived - reload the page.' });
      let docId = '';
      if (t === 'changes' && p.docId) {                      // which reviewer doc this person finished
        docId = String(p.docId);
        const ok = vs[vs.length - 1].doc === docId || vrRows_('VideoDocs').some(function (d) { return d.key === key && Number(d.v) === v && d.docId === docId; });
        if (!ok) return vrJson_({ ok: false, code: 'bad_doc', message: 'That document does not belong to this version.' });
        if (vrRows_('VideoEvents').some(function (e) { return e.key === key && Number(e.v) === v && e.type === 'changes' && e.docId === docId; })) return vrJson_({ ok: false, code: 'already', message: 'This document is already marked done.' });
      }
      vrTab_('VideoEvents', true).appendRow([key, v, t, by, vrNow_(), docId]);
      return vrJson_({ ok: true });
    }
    if (p.action === 'v_newdoc') {                           // "Open my comments": the reviewer's own doc for this version (idempotent per name)
      const key = String(p.key), v = Number(p.v), vs = vrVersions_()[key] || [];
      if (!vs.length || v !== vs[vs.length - 1].v) return vrJson_({ ok: false, code: 'stale', message: 'A newer version arrived - reload the page.' });
      const rows = vrRows_('VideoDocs').filter(function (d) { return d.key === key && Number(d.v) === v && d.by !== 'ALL'; });
      const mine = rows.filter(function (d) { return String(d.by).toLowerCase() === by.toLowerCase(); })[0];
      if (mine) return vrJson_({ ok: true, docId: mine.docId });          // this name already has a doc: show only that one
      const first = vs[vs.length - 1].doc, firstTaken = rows.some(function (d) { return d.docId === first; });
      let docId = first;                                                  // the first reviewer takes the doc made at ingest, the next ones get new blank docs
      if (!first || firstTaken) docId = vrMakeDoc_(key, v, '', by);
      vrTab_('VideoDocs', true).appendRow([key, v, docId, by, vrNow_()]); vrLog_('reviewer_doc', key, docId, by); return vrJson_({ ok: true, docId: docId });
    }
    if (p.action === 'v_note') {
      if (!/^[^|]+\|[^|]+\|[A-Z]{1,3}\d{2}$/.test(String(p.key))) return vrJson_({ ok: false, code: 'bad_key' });
      const text = String(p.text || '').trim().slice(0, 2000), tc = Number(p.tc), v = Number(p.v);
      if (!text || !(tc >= 0) || !(v >= 1)) return vrJson_({ ok: false, code: 'bad_note', message: 'Write the change first.' });
      let shotId = '';
      if (p.image) {
        const m = /^data:image\/jpeg;base64,([A-Za-z0-9+\/=]+)$/.exec(p.image);
        if (!m || m[1].length > 3000000) return vrJson_({ ok: false, code: 'bad_image', message: 'Screenshot was empty - save without it or try again.' });
        shotId = vrShots_().createFile(Utilities.newBlob(Utilities.base64Decode(m[1]), 'image/jpeg', p.key.replace(/[|\s]/g, '_') + '_v' + v + '_' + Utilities.getUuid().slice(0, 6) + '.jpg')).getId();
      }
      const nid = 'n' + Utilities.getUuid().replace(/-/g, '').slice(0, 8);     // letter prefix: Sheets can never read it as a number
      vrTab_('VideoNotes', true).appendRow([nid, p.key, v, tc, String(p.tcText || ''), text, by, vrNow_(), shotId, '', '', '']);
      return vrJson_({ ok: true, id: nid, shotId: shotId });
    }
    if (p.action === 'v_fixed') {
      const r = vrRows_('VideoNotes').filter(function (n) { return n.id === String(p.id); })[0];
      if (!r) return vrJson_({ ok: false, code: 'bad_note' });
      vrTab_('VideoNotes', true).getRange(r._row, 10, 1, 3).setValues([[p.on ? 'yes' : '', p.on ? by : '', p.on ? vrNow_() : '']]);
      return vrJson_({ ok: true });
    }
    return vrJson_({ ok: false, code: 'bad_action' });
  } catch (err) { return vrJson_({ ok: false, code: 'write_failed', message: String(err) }); }
}
/** PIN "Approve": COPY <inbox>/<PILLAR>/<LANG>/<CODE>.mp4 -> NEW NOW MATERIALS/<PILLAR>/VIDEO/<LANG>/<CODE>.mp4 (existing final -> that folder's _old/). */
function vrApprove_(key, v) {
  const a = key.split('|'); if (a.length !== 3 || VR_PILLARS.indexOf(a[0]) < 0) return vrJson_({ ok: false, code: 'bad_key' });
  const vs = vrVersions_()[key] || [], last = vs[vs.length - 1];
  if (!last || last.v !== v) return vrJson_({ ok: false, code: 'stale', message: 'A newer version arrived - reload the page.' });
  const ev = vrRows_('VideoEvents').filter(function (e) { return e.key === key; }).pop();
  const need = a[1] === 'ENGLISH' ? 'farah_ok' : 'ok';       // English: reviewers ok -> Farah ok -> Sarosh PIN (last step)
  if (!ev || ev.type !== need || Number(ev.v) !== v) return vrJson_({ ok: false, code: 'not_ok', message: a[1] === 'ENGLISH' ? 'Farah has not approved this version.' : 'The reviewer has not approved this version.' });
  const inbox = vrInbox_(), pf = inbox && vrSub_(inbox, a[0], false), L = pf && vrSub_(pf, a[1], false);
  const cur = L && L.getFilesByName(a[2] + '.mp4'); if (!cur || !cur.hasNext()) return vrJson_({ ok: false, code: 'no_file', message: a[2] + '.mp4 is not in the review folder.' });
  const file = cur.next(); if (file.getId() !== last.id) return vrJson_({ ok: false, code: 'stale', message: 'The file in the folder is not the reviewed version.' });
  const root = DriveApp.getFolderById(VR_ROOT), pfin = vrSub_(root, a[0], false), vf = pfin && vrSub_(pfin, 'VIDEO', false), dest = vf && vrSub_(vf, a[1], false);
  if (!dest) return vrJson_({ ok: false, code: 'no_dest', message: 'Final folder ' + a[0] + '/VIDEO/' + a[1] + ' not found.' });
  const ex = dest.getFilesByName(a[2] + '.mp4'), olds = [];
  while (ex.hasNext()) olds.push(ex.next());
  const moved = [];
  olds.forEach(function (o) {                              // existing final -> _old (dated), never overwritten
    const od = vrSub_(dest, '_old', true), to = a[2] + '_final_' + Utilities.formatDate(o.getLastUpdated(), 'Etc/UTC', 'yyyy-MM-dd_HHmm') + '.mp4', orig = o.getName();
    vrMoveInto_(o, od, to); moved.push({ f: o, name: orig }); vrLog_('final_to_old', key, o.getId(), to);
  });
  let copy;
  try { copy = file.makeCopy(a[2] + '.mp4', dest); }
  catch (err) {                                            // copy failed: put the old final back so the folder is never left empty
    moved.forEach(function (m) { try { m.f.moveTo(dest); m.f.setName(m.name); } catch (e2) { vrLog_('restore_failed', key, m.f.getId(), String(e2)); } });
    vrLog_('approve_failed', key, file.getId(), String(err)); return vrJson_({ ok: false, code: 'copy_failed', message: 'Copy failed, nothing changed: ' + err });
  }
  vrTab_('VideoEvents', true).appendRow([key, v, 'approved', 'Sarosh', vrNow_()]); vrLog_('approved_copy', key, copy.getId(), 'copied to ' + a[0] + '/VIDEO/' + a[1]);
  CacheService.getScriptCache().remove('vr_finals');
  return vrJson_({ ok: true });
}

/** one-off: empty the body of every existing review doc (user 21 Oct: blank docs) */
function vrBlankExistingDocs() {
  vrRows_('VideoVersions').forEach(function (r) { if (!r.docId) return; try { const d = DocumentApp.openById(r.docId); d.getBody().clear(); d.saveAndClose(); } catch (err) { vrLog_('blank_failed', r.key, r.docId, String(err)); } });
}
